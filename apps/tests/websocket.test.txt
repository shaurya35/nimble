import { describe, it, expect } from "bun:test";
import { WebSocket as WSClient } from "ws";

const WS_URL = "ws://localhost:8000";

const connectWebSocket = (): Promise<WSClient> => {
  return new Promise((resolve, reject) => {
    const ws = new WSClient(WS_URL);
    ws.on("open", () => resolve(ws));
    ws.on("error", reject);
  });
};

const sendMessage = (ws: WSClient, message: any): Promise<any> => {
  return new Promise((resolve, reject) => {
    const timeout = setTimeout(() => {
      reject(new Error("Timeout waiting for response"));
    }, 30000);

    ws.once("message", (data) => {
      clearTimeout(timeout);
      try {
        const response = JSON.parse(data.toString());
        resolve(response);
      } catch (error) {
        reject(error);
      }
    });

    ws.send(JSON.stringify(message));
  });
};

describe("WebSocket Server End-to-End Tests", () => {
  describe("Connection", () => {
    it("should connect to WebSocket server at port 8000", async () => {
      const ws = await connectWebSocket();
      expect(ws.readyState).toBe(WSClient.OPEN);
      ws.close();
    });
  });

  describe("Message Validation", () => {
    it("should reject invalid JSON", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, "invalid json");
      expect(response.success).toBe(false);
      expect(response.error).toContain("Invalid JSON format");
      ws.close();
    });

    it("should reject message with missing required fields", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        message: "test",
      });
      expect(response.success).toBe(false);
      expect(response.error).toBe("Invalid Configuration!");
      ws.close();
    });

    it("should reject message with invalid field types", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        message: 123,
        apiKey: "test",
        model: "test",
        config: {},
      });
      expect(response.success).toBe(false);
      expect(response.error).toBe("Invalid Configuration!");
      ws.close();
    });
  });

  describe("Create Todo Operations", () => {
    const validMessage = {
      message: "Add buy groceries",
      apiKey: process.env.GEMINI_API_KEY,
      model: "gemini-2.5-flash",
      config: {},
      existingTodos: "[]",
    };

    it("should create a simple todo", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, validMessage);

      expect(response.success).toBe(true);
      expect(response.operation).toBeDefined();
      expect(response.operation.type).toBe("create");
      expect(response.command).toBeDefined();
      expect(response.command).toContain("localStorage");
      expect(response.command).toContain("todos");
      ws.close();
    });

    it("should create todo with tomorrow date", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message: "Add complete assignments tomorrow",
      });

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("create");
      expect(response.operation.data).toBeDefined();
      expect(response.command).toContain("dueDate");
      ws.close();
    });

    it("should create todo with priority", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message: "High priority: fix the bug",
      });

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("create");
      expect(response.operation.data?.priority).toBe("high");
      ws.close();
    });

    it("should create todo in X days", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message: "Add review code in 3 days",
      });

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("create");
      expect(response.operation.data?.dueDate).toBeDefined();
      expect(response.command).toContain("dueDate");
      ws.close();
    });

    it("should create todo for next week", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message: "Remind me to call mom next week",
      });

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("create");
      expect(response.operation.data?.dueDate).toBeDefined();
      ws.close();
    });
  });

  describe("Update Todo Operations", () => {
    const existingTodos = JSON.stringify([
      { id: "123", title: "buy groceries", completed: false },
    ]);

    const validMessage = {
      message: "Mark buy groceries as done",
      apiKey: process.env.GEMINI_API_KEY || "",
      model: "gemini-2.5-flash",
      config: {},
      existingTodos,
    };

    it("should update todo completion status", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, validMessage);

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("update");
      expect(response.operation.data?.completed).toBe(true);
      expect(response.command).toContain("completed");
      ws.close();
    });

    it("should update todo priority", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message: "Change priority of buy groceries to high",
      });

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("update");
      ws.close();
    });
  });

  describe("Delete Todo Operations", () => {
    const existingTodos = JSON.stringify([
      { id: "123", title: "buy groceries", completed: false },
    ]);

    const validMessage = {
      message: "Delete buy groceries",
      apiKey: process.env.GEMINI_API_KEY || "",
      model: "gemini-2.5-flash",
      config: {},
      existingTodos,
    };

    it("should delete a todo", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, validMessage);

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("remove");
      expect(response.command).toContain("filter");
      ws.close();
    });

    it("should clear all todos", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message: "Clear all todos",
      });

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("clear");
      expect(response.command).toContain("localStorage.setItem");
      expect(response.command).toContain("JSON.stringify([])");
      ws.close();
    });
  });

  describe("List Todo Operations", () => {
    const existingTodos = JSON.stringify([
      { id: "123", title: "buy groceries", completed: false },
      { id: "456", title: "workout", completed: true },
    ]);

    const validMessage = {
      message: "Show my todos",
      apiKey: process.env.GEMINI_API_KEY || "",
      model: "gemini-2.5-flash",
      config: {},
      existingTodos,
    };

    it("should list todos", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, validMessage);

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("list");
      ws.close();
    });
  });

  describe("Date Calculation", () => {
    const validMessage = {
      message: "",
      apiKey: process.env.GEMINI_API_KEY || "",
      model: "gemini-2.5-flash",
      config: {},
      existingTodos: "[]",
    };

    it("should handle tomorrow dates correctly", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message: "Add task tomorrow",
      });

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("create");
      expect(response.operation.data?.dueDate).toBeDefined();
      expect(typeof response.operation.data?.dueDate).toBe("string");
      expect(response.operation.data?.dueDate).toMatch(/^\d{4}-\d{2}-\d{2}T00:00:00\.000Z$/);
      ws.close();
    });

    it("should handle 'in X days' dates correctly", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message: "Add review in 5 days",
      });

      expect(response.success).toBe(true);
      expect(response.operation.data?.dueDate).toBeDefined();
      expect(response.operation.data?.dueDate).toMatch(/^\d{4}-\d{2}-\d{2}T00:00:00\.000Z$/);
      ws.close();
    });

    it("should handle next week dates correctly", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message: "Add call mom next week",
      });

      expect(response.success).toBe(true);
      expect(response.operation.data?.dueDate).toBeDefined();
      expect(response.operation.data?.dueDate).toMatch(/^\d{4}-\d{2}-\d{2}T00:00:00\.000Z$/);
      ws.close();
    });
  });

  describe("Complex Scenarios", () => {
    const validMessage = {
      message: "",
      apiKey: process.env.GEMINI_API_KEY || "",
      model: "gemini-2.5-flash",
      config: {},
      existingTodos: "[]",
    };

    it("should handle multiple todos in existingTodos", async () => {
      const existingTodos = JSON.stringify([
        { id: "1", title: "todo 1", completed: false },
        { id: "2", title: "todo 2", completed: false },
        { id: "3", title: "todo 3", completed: true },
      ]);

      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message: "Mark todo 2 as done",
        existingTodos,
      });

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("update");
      ws.close();
    });

    it("should handle complex todo creation", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, {
        ...validMessage,
        message:
          "Add a task to complete assignments for SEA and complete projects tomorrow",
      });

      expect(response.success).toBe(true);
      expect(response.operation.type).toBe("create");
      expect(response.operation.data?.title).toBeDefined();
      expect(response.operation.data?.dueDate).toBeDefined();
      ws.close();
    });
  });

  describe("Response Format", () => {
    const validMessage = {
      message: "Add test todo",
      apiKey: process.env.GEMINI_API_KEY || "",
      model: "gemini-2.5-flash",
      config: {},
      existingTodos: "[]",
    };

    it("should return correct response structure", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, validMessage);

      expect(response).toHaveProperty("success");
      expect(response).toHaveProperty("command");
      expect(response).toHaveProperty("operation");
      expect(response.operation).toHaveProperty("type");
      expect(response.operation).toHaveProperty("localStorageCommand");
      ws.close();
    });

    it("should include command in response", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, validMessage);

      expect(response.command).toBeDefined();
      expect(typeof response.command).toBe("string");
      expect(response.command.length).toBeGreaterThan(0);
      ws.close();
    });

    it("should return executable localStorage command", async () => {
      const ws = await connectWebSocket();
      const response = await sendMessage(ws, validMessage);

      expect(response.command).toContain("localStorage");
      expect(response.command).toContain("setItem");
      expect(response.command).toContain("todos");
      ws.close();
    });
  });

  describe("Error Handling", () => {
    it("should handle invalid API key gracefully", async () => {
      const ws = await connectWebSocket();
      const invalidApiKeyMessage = {
        message: "Add test",
        apiKey: "invalid-key",
        model: "gemini-2.5-flash",
        config: {},
        existingTodos: "[]",
      };

      try {
        const response = await sendMessage(ws, invalidApiKeyMessage);
        if (!response.success) {
          expect(response.error).toBeDefined();
        }
      } catch (error) {
        expect(error).toBeDefined();
      }
      ws.close();
    });
  });
});
